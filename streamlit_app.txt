# ── Scrape Pipeline ───────────────────────────────────────────────────────────
def colorize(line):
    l = line.strip()
    if not l: return ""
    if "✓" in l or "done" in l.lower() or "complete" in l.lower() or "saved" in l.lower():
        return f'<div class="terminal-line t-green">✔ {l}</div>'
    elif "✗" in l or "failed" in l.lower() or "error" in l.lower():
        return f'<div class="terminal-line t-red">✗ {l}</div>'
    elif l.startswith("[") and "]" in l:
        tag = l[l.index("[")+1:l.index("]")]
        rest = l[l.index("]")+1:]
        color_map = {"Internshala":"t-green","LinkedIn":"t-blue",
                     "Naukri":"t-yellow","Export":"t-white",
                     "CSV":"t-green","XLSX":"t-green",
                     "Filter":"t-yellow","Groq":"t-blue",
                     "LLM":"t-blue","Embedder":"t-blue",
                     "Clusterer":"t-yellow","Analyzer":"t-yellow"}
        c = color_map.get(tag,"t-gray")
        return f'<div class="terminal-line"><span class="{c}">[{tag}]</span>{rest}</div>'
    elif l.startswith("="):
        return f'<div class="terminal-line t-gray">{l}</div>'
    elif "Page" in l or "Offset" in l or "Total" in l:
        return f'<div class="terminal-line t-gray" style="padding-left:16px">{l}</div>'
    else:
        return f'<div class="terminal-line t-white">{l}</div>'

    progress = st.progress(0)
    progress_text = st.empty()
    STEPS = [
        ("main.py",  "🕷️ Scraping Platforms",     10, 45),
        ("main2.py", "🧠 LLM Skill Extraction",   45, 78),
        ("main3.py", "🗺️ Embeddings + Clustering", 78, 92),
        ("main4.py", "📊 Building Profiles",       92,100),
    ]
    pipeline_failed = False
    for cmd, label, p_start, p_end in STEPS:
        progress.progress(p_start)
        progress_text.markdown(f"🟢 **{p_start}%** — {label}...")
        with st.expander(f"{label}", expanded=True):
            out_container = st.empty()
            lines = []
            proc = subprocess.Popen(
                [venv_python, cmd],
                stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                text=True, cwd=PROJECT_DIR, bufsize=1
            )
            for line in proc.stdout:
                l = line.rstrip()
                if l:
                    lines.append(colorize(l))
                    html = "".join(lines[-25:])
                    out_container.markdown(
                        f'<div style="background:#070f1c;padding:12px;border-radius:8px;'
                        f'border:1px solid rgba(46,134,171,0.2);max-height:300px;overflow-y:auto">'
                        f'{html}</div>', unsafe_allow_html=True)
            proc.wait()
        progress.progress(p_end)
        progress_text.markdown(f"🟢 **{p_end}%** — ✅ {label} done!")
        if proc.returncode != 0:
            st.error(f"❌ {label} failed! Check output above.")
            pipeline_failed = True
            break
    if not pipeline_failed:

# ── Dynamic Phase Writing Background Scraper Engine ───────────────────────────
def _start_background_scrape(role: str, location: str = "India") -> bool:
    if os.path.exists(_BG_RUNNING):
        return False
    os.makedirs("data", exist_ok=True)
    if os.path.exists(_BG_DONE): os.remove(_BG_DONE)
    if os.path.exists(_BG_STOP): os.remove(_BG_STOP)
        
    with open("data/scrape_config_override.json", "w") as f:
        json.dump({"roles": [role], "max_jobs": 30, "locations": [location]}, f)
    with open(_BG_RUNNING, "w") as f:
        f.write(role)
        
    PROJECT_DIR = os.path.dirname(os.path.abspath(__file__))
    runner_code = (
        f"import subprocess, sys, os\n"
        f"cwd = {repr(PROJECT_DIR)}\n"
        f"phases = [\n"
        f"    ('main.py', '🕷️ Scraping LinkedIn · Naukri · Internshala'),\n"
        f"    ('main2.py', '🧠 LLM skill extraction (Groq)'),\n"
        f"    ('main3.py', '🗺️ Embedding & clustering'),\n"
        f"    ('main4.py', '📊 Building market profiles')\n"
        f"]\n"
        f"broken = False\n"
        f"for script, phase_text in phases:\n"
        f"    if os.path.exists({repr(_BG_STOP)}): broken = True; break\n"
        f"    with open({repr(_BG_PHASE)}, 'w') as pf: pf.write(phase_text)\n"
        f"    r = subprocess.run([sys.executable, script], capture_output=True, cwd=cwd)\n"
        f"    if r.returncode != 0: broken = True; break\n"
        f"if not broken and not os.path.exists({repr(_BG_STOP)}):\n"
        f"    open({repr(_BG_DONE)}, 'w').write('done')\n"
        f"for f in [{repr(_BG_RUNNING)}, {repr(_BG_PHASE)}, {repr(_BG_STOP)}]:\n"
        f"    try: os.remove(f)\n"
        f"    except: pass\n"
    )
    with open("data/bg_runner.py", "w") as f:
        f.write(runner_code)
    subprocess.Popen([sys.executable, "data/bg_runner.py"],
                     stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
                     cwd=PROJECT_DIR)
    return True

_KNOWN_ROLES = [
    "data analyst","data scientist","data engineer","business analyst",
    "software engineer","software developer","frontend developer",
    "backend developer","full stack developer","flutter developer",
    "react developer","angular developer","python developer","java developer",
    "machine learning engineer","ml engineer","ai engineer","product manager",
    "ux designer","ui designer","devops engineer","cloud engineer",
    "qa engineer","mobile developer","ios developer","android developer",
    "content writer","digital marketer","project manager","blockchain developer",
    "lawyer","legal associate",
]

def _detect_role_quick(text: str):
    tl = text.lower()
    for r in _KNOWN_ROLES:
        if r in tl:
            return r.title()
    m = re.search(
        r'\b([a-z]+(?:\s+[a-z]+)?)\s+'
        r'(developer|analyst|engineer|manager|designer|scientist|architect)\b', tl)
    if m and m.group(1) not in ("a","the","any","software","i"):
        return (m.group(1) + " " + m.group(2)).title()
    m2 = re.search(r'(?:analyze|search|about|find|for)\s+([a-z]+(?:\s+[a-z]+)?)\s+jobs?', tl)
    if m2:
        r = m2.group(1).strip()
        if 1 <= len(r.split()) <= 3:
            return r.title()
    return None

def _extract_skills_from_history(chat_msgs: list) -> list[str]:
    user_msgs = [m["content"] for m in chat_msgs if m["role"] == "user"]
    for msg in reversed(user_msgs):
        if "," in msg and len(msg) < 300:
            parts = [p.strip() for p in msg.split(",") if p.strip()]
            if 2 <= len(parts) <= 20:
                return parts
    return []

def _render_gap_chart(gap_data: dict):
    ranked   = dict(gap_data.get("all_ranked", []))
    priority = gap_data.get("priority_list", [])
    missing  = set(gap_data.get("missing_required", []))
    if not priority: return
    rows = [{"Skill": s, "% in JDs": ranked.get(s, 0),
             "Type": "🔴 Must Learn" if s in missing else "🟡 Useful"}
            for s in priority[:12]]
    df_chart = pd.DataFrame(rows)
    fig = px.bar(df_chart, x="% in JDs", y="Skill", orientation="h", color="Type",
                 color_discrete_map={"🔴 Must Learn":"#FF4444","🟡 Useful":"#FFB347"},
                 title=f"Learning Roadmap — Readiness: {gap_data.get('readiness_score',0)}%",
                 text="% in JDs")
    fig.update_traces(texttemplate="%{text}%", textposition="outside")
    fig.update_layout(height=max(300, len(rows)*35),
                      yaxis={"categoryorder":"total ascending"},
                      plot_bgcolor="#070f1c", paper_bgcolor="#070f1c",
                      font_color="white")
    st.plotly_chart(fig, use_container_width=True)
